find_package(FlatBuffers CONFIG REQUIRED)
find_path(MCAP_INCLUDE_DIR mcap/writer.hpp REQUIRED)
set(schema_dir ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${schema_dir})
foreach(schema system navigation)
  if(schema STREQUAL system)
    set(source ${CMAKE_CURRENT_SOURCE_DIR}/../runtime/schema/system.fbs)
  else()
    set(source ${CMAKE_CURRENT_SOURCE_DIR}/../ins/schema/navigation.fbs)
  endif()
  add_custom_command(OUTPUT ${schema_dir}/${schema}_generated.h ${schema_dir}/${schema}.bfbs
    COMMAND flatbuffers::flatc --cpp --scoped-enums -o ${schema_dir} ${source}
    COMMAND flatbuffers::flatc -b --schema -o ${schema_dir} ${source}
    DEPENDS ${source} VERBATIM)
endforeach()
add_library(pl64-service-common STATIC mcap.cpp ${schema_dir}/system_generated.h ${schema_dir}/navigation_generated.h)
target_include_directories(pl64-service-common PUBLIC ${schema_dir} ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(pl64-service-common SYSTEM PRIVATE ${MCAP_INCLUDE_DIR})
target_link_libraries(pl64-service-common PUBLIC pl64::runtime flatbuffers::flatbuffers pl64-mcap-crc)
target_compile_definitions(pl64-service-common PRIVATE MCAP_COMPRESSION_NO_LZ4 MCAP_COMPRESSION_NO_ZSTD)
target_compile_options(pl64-service-common PRIVATE -Wall -Wextra -Wpedantic)
foreach(tool recorder sar ctl)
  add_executable(pl64-${tool} ${tool}.cpp)
  target_link_libraries(pl64-${tool} PRIVATE pl64-service-common)
  target_compile_options(pl64-${tool} PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(pl64-${tool} PRIVATE PL64_SYSTEM_SCHEMA="${CMAKE_INSTALL_PREFIX}/share/pl64/system.bfbs")
  install(TARGETS pl64-${tool} RUNTIME DESTINATION bin)
endforeach()
add_executable(pl64-mcap-check check.cpp)
target_link_libraries(pl64-mcap-check PRIVATE pl64-service-common)
target_include_directories(pl64-mcap-check SYSTEM PRIVATE ${MCAP_INCLUDE_DIR})
target_compile_definitions(pl64-mcap-check PRIVATE MCAP_COMPRESSION_NO_LZ4 MCAP_COMPRESSION_NO_ZSTD)
install(TARGETS pl64-mcap-check RUNTIME DESTINATION bin)
install(FILES ${schema_dir}/system.bfbs ../runtime/schema/system.fbs DESTINATION share/pl64)
if(PL64_BUILD_GUI)
  include(gui.cmake)
endif()
if(BUILD_TESTING)
  add_executable(pl64-test-source test_source.cpp)
  target_link_libraries(pl64-test-source PRIVATE pl64-service-common)
  add_executable(service-tests tests.cpp)
  target_link_libraries(service-tests PRIVATE pl64-service-common)
  target_include_directories(service-tests SYSTEM PRIVATE ${MCAP_INCLUDE_DIR})
  target_compile_definitions(service-tests PRIVATE MCAP_COMPRESSION_NO_LZ4 MCAP_COMPRESSION_NO_ZSTD
    PL64_TEST_SCHEMA="${schema_dir}/system.bfbs")
  add_test(NAME system-recording COMMAND service-tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME capture-control COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_capture_control.py
    $<TARGET_FILE:pl64-recorder> $<TARGET_FILE:pl64-test-source> ${schema_dir}/navigation.bfbs
    $<TARGET_FILE:pl64-mcap-check>)
endif()
