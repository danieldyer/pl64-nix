cmake_minimum_required(VERSION 3.25)
project(pl64_runtime VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(Threads REQUIRED)
add_library(pl64-runtime src/ipc.cpp src/control.cpp)
add_library(pl64::runtime ALIAS pl64-runtime)
target_compile_features(pl64-runtime PUBLIC cxx_std_23)
target_include_directories(pl64-runtime PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(pl64-runtime PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(pl64-runtime PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(pl64-runtime PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME runtime)
option(PL64_RUNTIME_INSTALL "Install the standalone runtime development package" ${PROJECT_IS_TOP_LEVEL})
if(PL64_RUNTIME_INSTALL)
  install(TARGETS pl64-runtime EXPORT pl64-runtime-targets)
  install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(FILES schema/system.fbs DESTINATION ${CMAKE_INSTALL_DATADIR}/pl64-runtime)
  install(EXPORT pl64-runtime-targets NAMESPACE pl64:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pl64-runtime)
  install(FILES cmake/pl64-runtime-config.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pl64-runtime)
endif()
if(BUILD_TESTING)
  add_executable(runtime-tests tests/runtime.cpp)
  target_link_libraries(runtime-tests PRIVATE pl64::runtime)
  add_test(NAME runtime-ipc COMMAND runtime-tests)
  set_tests_properties(runtime-ipc PROPERTIES TIMEOUT 30)
endif()
add_executable(runtime-benchmark EXCLUDE_FROM_ALL tests/benchmark.cpp)
target_link_libraries(runtime-benchmark PRIVATE pl64::runtime)
